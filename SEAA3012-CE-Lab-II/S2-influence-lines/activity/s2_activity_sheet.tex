\begin{enumerate}[label=\textbf{\arabic*.},itemsep=3pt]
\item \textbf{Read the reactions.} A unit load stands at halt 6, on the suspended span. Predict which of the six reactions are non-zero, and their signs.
\pbox{1.7cm}
Explorer (\emph{Quick fill: Unit load}, halt 6): $R_C = $ \underline{\hspace{1.6cm}}\quad $R_E = $ \underline{\hspace{1.6cm}}

\item \textbf{Open an explanation.} Tap the $R_E$ box. $\il{D} = $ \underline{\hspace{1.6cm}}. In one line: why is $\il{E} = 1.76\,\il{D}$?
\pbox{1.6cm}

\item \textbf{Set your own loads.} $P_1 = \SI{3}{kg}$, $P_2 = \SI{2}{kg}$, $P_3 = 0$, spacing \SI{140}{mm}, $P_1$ at halt B. Predict the sign of $R_A$ and why.
\pbox{1.6cm}
Explorer: $R_A = $ \underline{\hspace{1.6cm}} kgf\quad $R_B = $ \underline{\hspace{1.6cm}} kgf

\item \textbf{Find the critical position.} With the same loads, which load should stand on C for the largest $R_B$: the 3~kg or the 2~kg? Prediction and reason:
\pbox{1.6cm}
Explorer: largest $R_B = $ \underline{\hspace{1.6cm}} kgf with $P_1$ at halt \underline{\hspace{1cm}}. Was your prediction right? \underline{\hspace{3.6cm}}

\item \textbf{Use the areas.} Show only $R_E$ and shade the areas. Predict the sign of $R_E$ under a uniform load over the whole model.
\pbox{1.5cm}
Explorer: net area \underline{\hspace{1.6cm}} mm, so for $w = \SI{0.015}{kg/mm}$, $R_E = $ \underline{\hspace{1.6cm}} kgf.
\end{enumerate}
